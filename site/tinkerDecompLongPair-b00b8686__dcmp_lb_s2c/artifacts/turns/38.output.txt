def unproject_points(mask, depth, intrinsics):
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    zs = depth[ys, xs]
    # Filter out invalid depths (0 or nan)
    valid = zs > 0
    xs = xs[valid]
    ys = ys[valid]
    zs = zs[valid]
    if len(xs) == 0:

    fx = intrinsics[0, 0]

    cy = intrinsics[1, 2]
    xs_3d = (xs - cx) * zs / fx
    ys_3d = (ys - cy) * zs / fy
    zs_3d = zs
    points = np.column_stack([xs_3d, ys_3d, zs_3d])
    return points
