def mask_to_3d_points(mask, depth, intrinsics):
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    points = []
    for i in range(len(xs)):
        u, v = xs[i], ys[i]
        d = depth[v, u]
        if d <= 0 or np.isnan(d):
            continue
        fx = intrinsics[0, 0]
        fy = intrinsics[1, 1]
        cx = intrinsics[0, 2]
        cy = intrinsics[1, 2]
        x = (u - cx) * d / fx
        y = (v - cy) * d / fy
        z = d
        points.append([x, y, z])
    if not points:
        return None
    return np.array(points)
