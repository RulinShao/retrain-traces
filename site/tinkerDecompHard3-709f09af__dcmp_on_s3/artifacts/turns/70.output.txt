# Helper function to convert pixel to 3D point using depth
def pixel_to_3d(pixel_x, pixel_y, depth_img, intrinsics, pose_mat):
    """Convert pixel coordinates to 3D world coordinates."""
    z = depth_img[pixel_y, pixel_x]
    if z <= 0:
        return None
    
    # Camera frame coordinates
    x_cam = (pixel_x - intrinsics[0, 2]) * z / intrinsics[0, 0]
    y_cam = (pixel_y - intrinsics[1, 2]) * z / intrinsics[1, 1]
    z_cam = z
    
    # Transform to world frame using camera extrinsic matrix
    # pose_mat transforms from camera frame to world frame
    point_cam = np.array([x_cam, y_cam, z_cam, 1.0])
    point_world = pose_mat @ point_cam
    return point_world[:3]
