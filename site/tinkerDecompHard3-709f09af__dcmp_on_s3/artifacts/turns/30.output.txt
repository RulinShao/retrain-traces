def pixel_to_3d_camera(pixel, depth):
    if pixel is None or pixel[0] is None or pixel[1] is None:
        return None
    u, v = pixel
    if u < 0 or u >= depth.shape[1] or v < 0 or v >= depth.shape[0]:
        return None
    z = depth[v, u]
    if z <= 0 or np.isnan(z):
        return None
    x = (u - cx) * z / fx
    y = (v - cy) * z / fy
    return np.array([x, y, z])
