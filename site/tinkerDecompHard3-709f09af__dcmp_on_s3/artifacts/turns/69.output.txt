# Step 3: Get 3D points for both cubes
def get_3d_points_from_mask(mask, depth, intrinsics):
    """Get 3D points in camera frame from a binary mask and depth image."""
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    
    # Filter out invalid depth values
    valid = depth[ys, xs] > 0
    xs_valid = xs[valid]
    ys_valid = ys[valid]
    depths_valid = depth[ys_valid, xs_valid]
    
    if len(xs_valid) == 0:
        return None
    
    # Back-project to 3D in camera frame
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    
    z = depths_valid
    x = (xs_valid - cx) * z / fx
    y = (ys_valid - cy) * z / fy
    
    points_cam = np.column_stack([x, y, z])
    return points_cam

red_points_cam = get_3d_points_from_mask(red_mask, depth_image, intrinsics)

# Step 5: Transform to world frame
# The extrinsics matrix transforms from camera frame to world frame
# pose_mat is typically [R|t; 0|1] where R is 3x3 rotation and t is 3x1 translation

def transform_point_cam_to_world(point_cam, extrinsics):
    """Transform a point from camera frame to world frame."""
    point_hom = np.append(point_cam, 1.0)
    point_world = extrinsics @ point_hom
    return point_world[:3]

def transform_orientation_cam_to_world(R_cam, extrinsics):
    """Transform orientation from camera frame to world frame."""
    R_world_cam = extrinsics[:3, :3]
    R_world = R_world_cam @ R_cam
    return R_world
