def tree_sha(d):
    out = {}
    for r, _, fs in os.walk(d):
        for f in fs:
            p = os.path.join(r, f)
            if os.path.isfile(p) and "__pycache__" not in p and ".gw.sock" not in p and not f.endswith(".safetensors"):
                out[os.path.relpath(p, d)] = hashlib.sha256(open(p, "rb").read()).hexdigest()[:16]
    return dict(sorted(out.items()))

def child_hashes(scr, out):
    sys.path.insert(0, REPO)
    from retrain import session_config as SC
    D = {}
    # derived.probe_example (WL R-1017-AT(1)) is versioned explicitly: new configurations hash with it; the comparison with REV is
    # on the derived keys REV has (the comparison session_config create makes against an existing manifest)
    _sv = getattr(SC, "SEED_VERSION_KEYS", ("probe_example",))   # + practice_ctl_seed (10-18): seed versions added after REV
    _h = lambda cfg: SC.config_hash(dict(cfg, derived={k: v for k, v in cfg["derived"].items() if k not in _sv}))
    for name, (_, a, sw) in sorted(SC.INSTANCES.items()):
        cfg, errs = SC.build(dict(sw), backend="real", **a)
        D[name] = _h(cfg) if cfg else ["refused"] + errs
    open(out, "w").write(json.dumps(D, sort_keys=True))
