def pixel_to_3d(pixel_x, pixel_y, depth_map, intrinsics, pose_mat):
    """Convert pixel coordinates to 3D world coordinates"""
    if pixel_x is None or pixel_y is None:
        return None
    
    # Get depth at pixel location
    depth = depth_map[pixel_y, pixel_x]
    if depth <= 0 or np.isnan(depth):
        return None
    
    # Camera intrinsics
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    
    # Camera frame coordinates
    x_cam = (pixel_x - cx) * depth / fx
    y_cam = (pixel_y - cy) * depth / fy
    z_cam = depth
    
    # Transform from camera frame to world frame
    # pose_mat is camera extrinsic (camera to world)
    cam_to_world = pose_mat.copy()
    
    # Point in camera frame
    p_cam = np.array([x_cam, y_cam, z_cam, 1.0])
    
    # Transform to world frame
    p_world = cam_to_world @ p_cam
    
    return p_world[:3]
