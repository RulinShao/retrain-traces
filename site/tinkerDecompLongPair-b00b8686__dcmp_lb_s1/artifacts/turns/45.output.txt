def get_3d_points_from_mask(mask, depth, intrinsics, pose_mat):
    """Project masked pixels to 3D world coordinates."""
    mask = mask.astype(bool)
    ys, xs = np.where(mask)
    if len(xs) == 0:
        raise ValueError("No pixels in mask")
    
    # Get depth values for masked pixels
    z_vals = depth[ys, xs]
    valid = z_vals > 0
    xs = xs[valid]
    ys = ys[valid]
    z_vals = z_vals[valid]
    
    if len(xs) == 0:
        raise ValueError("No valid depth values in mask")
    
    # Back-project to camera frame
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    
    x_cam = (xs - cx) * z_vals / fx
    y_cam = (ys - cy) * z_vals / fy
    z_cam = z_vals
    
    points_cam = np.column_stack([x_cam, y_cam, z_cam])  # (N, 3)
    
    # Transform from camera frame to world frame
    # pose_mat is the camera extrinsic (4x4), which transforms camera frame to world frame
    ones = np.ones((len(points_cam), 1))
    points_h = np.hstack([points_cam, ones])  # (N, 4)
    points_world = (pose_mat @ points_h.T).T[:, :3]
    
    return points_world

def rotmat_to_quat_wxyz(R):
    """Convert 3x3 rotation matrix to quaternion [w, x, y, z]."""
    trace = np.trace(R)
    if trace > 0:
        S = np.sqrt(trace + 1.0) * 2
        w = 0.25 * S
        x = (R[2, 1] - R[1, 2]) / S
        y = (R[0, 2] - R[2, 0]) / S
        z = (R[1, 0] - R[0, 1]) / S
    elif (R[0, 0] > R[1, 1]) and (R[0, 0] > R[2, 2]):
        S = np.sqrt(1.0 + R[0, 0] - R[1, 1] - R[2, 2]) * 2
        w = (R[2, 1] - R[1, 2]) / S
        x = 0.25 * S
        y = (R[0, 1] + R[1, 0]) / S
        z = (R[0, 2] + R[2, 0]) / S
    elif R[1, 1] > R[2, 2]:
        S = np.sqrt(1.0 + R[1, 1] - R[0, 0] - R[2, 2]) * 2
        w = (R[0, 2] - R[2, 0]) / S
        x = (R[0, 1] + R[1, 0]) / S
        y = 0.25 * S
        z = (R[1, 2] + R[2, 1]) / S
    else:
        S = np.sqrt(1.0 + R[2, 2] - R[0, 0] - R[1, 1]) * 2
        w = (R[1, 0] - R[0, 1]) / S
        x = (R[0, 2] + R[2, 0]) / S
        y = (R[1, 2] + R[2, 1]) / S
        z = 0.25 * S
    
    quat = np.array([w, x, y, z])
    quat = quat / np.linalg.norm(quat)
    return quat
