        best_idx = np.argmax(grasp_scores)
        best_grasp = grasp_poses[best_idx]
        
        # Transform grasp from camera frame to world frame
        # best_grasp is in camera frame, we need to apply pose_mat to get world frame
        world_grasp = pose_mat @ best_grasp
        
        # Extract position and orientation
        target_pos = world_grasp[:3, 3]
        target_rot = world_grasp[:3, :3]
        
        # Convert rotation matrix to quaternion
        # Using the standard algorithm
        trace = np.trace(target_rot)
        if trace > 0:
            S = np.sqrt(trace + 1.0) * 2
            w = 0.25 * S
            x = (target_rot[2, 1] - target_rot[1, 2]) / S
            y = (target_rot[0, 2] - target_rot[2, 0]) / S
            z = (target_rot[1, 0] - target_rot[0, 1]) / S
        elif (target_rot[0, 0] > target_rot[1, 1]) and (target_rot[0, 0] > target_rot[2, 2]):
            S = np.sqrt(1.0 + target_rot[0, 0] - target_rot[1, 1] - target_rot[2, 2]) * 2
            w = (target_rot[2, 1] - target_rot[1, 2]) / S
            x = 0.25 * S
            y = (target_rot[0, 1] + target_rot[1, 0]) / S
            z = (target_rot[0, 2] + target_rot[2, 0]) / S
        elif target_rot[1, 1] > target_rot[2, 2]:
            S = np.sqrt(1.0 + target_rot[1, 1] - target_rot[0, 0] - target_rot[2, 2]) * 2
            w = (target_rot[0, 2] - target_rot[2, 0]) / S
            x = (target_rot[0, 1] + target_rot[1, 0]) / S
            y = 0.25 * S
            z = (target_rot[1, 2] + target_rot[2, 1]) / S
        else:
            S = np.sqrt(1.0 + target_rot[2, 2] - target_rot[0, 0] - target_rot[1, 1]) * 2

            x = (target_rot[0, 2] + target_rot[2, 0]) / S
            y = (target_rot[1, 2] + target_rot[2, 1]) / S
            z = 0.25 * S

        move_to_joints(lift_joints)

            ys, xs = np.where(stove_mask)

                cx = np.mean(xs)

                    cam_y = (cy - intrinsics[1, 2]) * depth_val / intrinsics[1, 1]
                    cam_z = depth_val
