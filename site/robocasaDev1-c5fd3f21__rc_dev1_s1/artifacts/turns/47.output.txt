# Convert rotation matrix to quaternion
def rotmat_to_quat(R):
    trace = np.trace(R)
    if trace > 0:
        S = np.sqrt(trace + 1.0) * 2
        w = 0.25 * S
        x = (R[2, 1] - R[1, 2]) / S
        y = (R[0, 2] - R[2, 0]) / S
        z = (R[1, 0] - R[0, 1]) / S
    elif (R[0, 0] > R[1, 1]) and (R[0, 0] > R[2, 2]):
        S = np.sqrt(1.0 + R[0, 0] - R[1, 1] - R[2, 2]) * 2
        w = (R[2, 1] - R[1, 2]) / S
        x = 0.25 * S
        y = (R[0, 1] + R[1, 0]) / S
        z = (R[0, 2] + R[2, 0]) / S
    elif R[1, 1] > R[2, 2]:
        S = np.sqrt(1.0 + R[1, 1] - R[0, 0] - R[2, 2]) * 2
        w = (R[0, 2] - R[2, 0]) / S
        x = (R[0, 1] + R[1, 0]) / S
        y = 0.25 * S
        z = (R[1, 2] + R[2, 1]) / S
    else:
        S = np.sqrt(1.0 + R[2, 2] - R[0, 0] - R[1, 1]) * 2
        w = (R[1, 0] - R[0, 1]) / S
        x = (R[0, 2] + R[2, 0]) / S
        y = (R[1, 2] + R[2, 1]) / S
        z = 0.25 * S
    quat = np.array([w, x, y, z])
    return quat / np.linalg.norm(quat)

quat_world = rotmat_to_quat(rot_world)

    # Find the center of the stove mask
    ys, xs = np.where(stove_mask)
    if len(xs) > 0:
        center_x = int(np.mean(xs))
        center_y = int(np.mean(ys))
        
        # Get depth at the center
        d = depth[center_y, center_x]
        if d > 0.05 and d < 3.0:
            # Convert to camera coordinates

            y_cam = (v - intrinsics[1, 2]) * d / intrinsics[1, 1]
            z_cam = d
            
            # Transform to world frame

            point_world = pose_mat @ point_cam
            stove_pos = point_world[:3]

            # Fallback: place at a reasonable position
