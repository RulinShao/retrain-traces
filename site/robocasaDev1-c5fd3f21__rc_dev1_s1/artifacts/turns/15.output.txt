def get_3d_points_from_mask(mask, depth, intrinsics, pose_mat):
    """Convert segmented mask to 3D points in robot base frame."""
    valid = (depth > 0.05) & (depth < 3.0) & mask
    if not np.any(valid):
        return None
    
    ys, xs = np.where(valid)
    if len(xs) == 0:
        return None
    
    # Subsample for efficiency
    if len(xs) > 1000:
        indices = np.random.choice(len(xs), 1000, replace=False)
        xs = xs[indices]
        ys = ys[indices]
    
    K = intrinsics
    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    
    depths = depth[ys, xs]
    points_cam = np.zeros((len(xs), 3))
    points_cam[:, 0] = (xs - cx) * depths / fx
    points_cam[:, 1] = (ys - cy) * depths / fy
    points_cam[:, 2] = depths
    
    # Transform to robot base frame
    points_world = np.zeros((len(xs), 4))
    points_world[:, :3] = points_cam
    points_world[:, 3] = 1.0
    points_world = (pose_mat @ points_world.T).T
    points_world = points_world[:, :3]
    
    return points_world

def get_object_center(points):
    if points is None or len(points) == 0:
        return None
    return np.mean(points, axis=0)

import numpy as np

def quat_from_matrix(R):
    """Convert rotation matrix to quaternion (w, x, y, z)."""
    trace = R[0, 0] + R[1, 1] + R[2, 2]
    if trace > 0:
        s = 0.5 / np.sqrt(trace + 1.0)
        w = 0.25 / s
        x = (R[2, 1] - R[1, 2]) * s
        y = (R[0, 2] - R[2, 0]) * s
        z = (R[1, 0] - R[0, 1]) * s
    elif R[0, 0] > R[1, 1] and R[0, 0] > R[2, 2]:
        s = 2.0 * np.sqrt(1.0 + R[0, 0] - R[1, 1] - R[2, 2])
        w = (R[2, 1] - R[1, 2]) / s
        x = 0.25 * s
        y = (R[0, 1] + R[1, 0]) / s
        z = (R[0, 2] + R[2, 0]) / s
    elif R[1, 1] > R[2, 2]:
        s = 2.0 * np.sqrt(1.0 + R[1, 1] - R[0, 0] - R[2, 2])
        w = (R[0, 2] - R[2, 0]) / s
        x = (R[0, 1] + R[1, 0]) / s
        y = 0.25 * s
        z = (R[1, 2] + R[2, 1]) / s
    else:
        s = 2.0 * np.sqrt(1.0 + R[2, 2] - R[0, 0] - R[1, 1])
        w = (R[1, 0] - R[0, 1]) / s
        x = (R[0, 2] + R[2, 0]) / s
        y = (R[1, 2] + R[2, 1]) / s
        z = 0.25 * s
    q = np.array([w, x, y, z])
    q = q / np.linalg.norm(q)
    if q[0] < 0:
        q = -q
    return q

def quat_multiply(q1, q2):
    """Multiply two quaternions (w, x, y, z)."""
    w1, x1, y1, z1 = q1
    w2, x2, y2, z2 = q2
    w = w1 * w2 - x1 * x2 - y1 * y2 - z1 * z2
    x = w1 * x2 + x1 * w2 + y1 * z2 - z1 * y2
    y = w1 * y2 - x1 * z2 + y1 * w2 + z1 * x2
    z = w1 * z2 + x1 * y2 - y1 * x2 + z1 * w2
    q = np.array([w, x, y, z])
    return q / np.linalg.norm(q)

def get_object_pose(rgb, depth, intrinsics, pose_mat, text_prompt):
    """Get the 3D pose of an object from segmentation."""
    masks = segment_sam3_text_prompt(rgb, text_prompt)
    if not masks:
        return None
    
    # Use the highest scoring mask
    best_mask = max(masks, key=lambda m: m.get('score', 0))
    mask = best_mask['mask']
    
    if not np.any(mask):
        return None
    
    # Create segmentation map (1 for the object)
    seg_map = mask.astype(np.int32)
    
    # Plan grasp
    grasp_poses, grasp_scores = plan_grasp(depth, intrinsics, seg_map)
    
    if len(grasp_poses) == 0:
        # Fall back to using the centroid of the mask
        ys, xs = np.where(mask)
        if len(xs) == 0:
            return None
        u = np.mean(xs)
        v = np.mean(ys)
        d = depth[v.astype(int), u.astype(int)]
        if d <= 0.05 or d > 3 or np.isnan(d):
            return None
        # Camera coordinates
        x_cam = (u - intrinsics[0, 2]) * d / intrinsics[0, 0]
        y_cam = (v - intrinsics[1, 2]) * d / intrinsics[1, 1]
        z_cam = d
        # Transform to world
        cam_point = np.array([x_cam, y_cam, z_cam, 1.0])
        world_point = pose_mat @ cam_point
        position = world_point[:3]
        # Use a default orientation (fingers down)
        quaternion = np.array([0.0, 0.0, 1.0, 0.0])
        return position, quaternion
    
    # Use the best grasp
    best_idx = np.argmax(grasp_scores)
    grasp_pose = grasp_poses[best_idx]
    
    # Transform from camera frame to world frame
    world_pose = pose_mat @ grasp_pose
    position = world_pose[:3, 3]
    R = world_pose[:3, :3]
    quaternion = quat_from_matrix(R)
    
    return position, quaternion
