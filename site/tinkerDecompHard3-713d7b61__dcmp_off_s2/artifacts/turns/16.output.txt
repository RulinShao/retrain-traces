def get_3d_points_from_mask(depth, intrinsics, pose_mat, mask):
    """
    Get 3D points in world frame from a depth image, intrinsics, extrinsics, and mask.
    """
    h, w = depth.shape
    ys, xs = np.where(mask)

        raise ValueError("Empty mask")

    return points_world
