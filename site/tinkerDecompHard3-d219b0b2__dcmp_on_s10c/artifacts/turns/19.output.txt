# Step 3: Get 3D points for green and red cubes from depth image
def mask_to_3d_points(mask, depth, intrinsics):
    """Convert a binary mask and depth image to 3D points in camera frame."""
    H, W = mask.shape
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return np.empty((0, 3))
    
    # Filter out invalid depth values
    valid = depth[ys, xs] > 0
    xs = xs[valid]
    ys = ys[valid]
    zs = depth[ys, xs]
    
    if len(xs) == 0:
        return np.empty((0, 3))
    
    # Convert pixel coordinates to 3D camera frame
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    
    x_cam = (xs - cx) * zs / fx
    y_cam = (ys - cy) * zs / fy
    z_cam = zs
    
    points = np.column_stack([x_cam, y_cam, z_cam])
    return points

def transform_points(points_cam, pose_mat):
    """Transform points from camera frame to world frame using extrinsic matrix."""
    # pose_mat is 4x4, points are in camera frame
    # World point = pose_mat @ [cam_point; 1]
    N = len(points_cam)
    ones = np.ones((N, 1))
    points_h = np.hstack([points_cam, ones])  # (N, 4)
    points_world = (pose_mat @ points_h.T).T[:, :3]
    return points_world
