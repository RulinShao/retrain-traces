def get_3d_points_from_mask(mask, depth, intrinsics, pose_mat):
    # Get pixel coordinates where mask is true
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    
    # Get depth values at those pixels
    depths = depth[ys, xs]
    
    # Filter out invalid depth values
    valid = np.isfinite(depths) & (depths > 0)
    xs = xs[valid]
    ys = ys[valid]
    depths = depths[valid]
    
    if len(xs) == 0:
        return None
    
    # Convert pixel coordinates to camera coordinates
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    
    # Camera coordinates
    z_cam = depths
    x_cam = (xs - cx) * z_cam / fx
    y_cam = (ys - cy) * z_cam / fy
    
    # Stack as (N, 3) array in camera frame
    points_cam = np.column_stack([x_cam, y_cam, z_cam])

    R = pose_mat[:3, :3]
    t = pose_mat[:3, 3]

    return points_world
