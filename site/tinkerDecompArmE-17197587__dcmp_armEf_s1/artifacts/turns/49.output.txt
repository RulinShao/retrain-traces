def mask_to_3d_points(mask, depth, intrinsics):
    """Convert a binary mask + depth image to 3D points in camera frame."""
    H, W = depth.shape
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return np.empty((0, 3))
    fx = intrinsics[0, 0]
    fy = intrinsics[1, 1]
    cx = intrinsics[0, 2]
    cy = intrinsics[1, 2]
    z_vals = depth[ys, xs]
    # Filter out invalid depth
    valid = z_vals > 0
    xs = xs[valid]
    ys = ys[valid]
    z_vals = z_vals[valid]
    x_cam = (xs - cx) * z_vals / fx
    y_cam = (ys - cy) * z_vals / fy
    points = np.stack([x_cam, y_cam, z_vals], axis=1)
    return points
