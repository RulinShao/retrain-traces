def get_3d_points_from_depth(depth, intrinsics, segmentation):
    """Get 3D points from depth image and segmentation mask"""

    """Convert rotation matrix to quaternion (w, x, y, z)"""

        w = (R[0, 2] - R[2, 0]) / S
        x = (R[0, 1] + R[1, 0]) / S
