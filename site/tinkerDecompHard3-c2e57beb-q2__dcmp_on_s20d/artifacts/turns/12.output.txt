def mask_to_3d_points(mask, depth, intrinsics):
    """Convert mask to 3D points in camera frame"""
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    
    # Get valid pixels with depth
    valid = depth[ys, xs] > 0
    xs_valid = xs[valid]
    ys_valid = ys[valid]
    depths_valid = depth[ys_valid, xs_valid]
    
    if len(xs_valid) == 0:

    x = (xs_valid - cx) * z / fx
    y = (ys_valid - cy) * z / fy

    return points_cam
