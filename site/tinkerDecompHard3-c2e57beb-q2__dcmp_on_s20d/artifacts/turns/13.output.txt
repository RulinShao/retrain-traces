        y = (R[1, 2] + R[2, 1]) / S
        z = 0.25 * S
    quat = np.array([w, x, y, z])

def get_3d_points_from_mask(mask, depth, intrinsics, pose_mat):
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    depths = depth[ys, xs]
    valid = depths > 0
    xs_valid = xs[valid]
    ys_valid = ys[valid]
    depths_valid = depths[valid]
    
    # Convert pixel coordinates to camera frame
    x_cam = (xs_valid - intrinsics[0, 2]) * depths_valid / intrinsics[0, 0]
    y_cam = (ys_valid - intrinsics[1, 2]) * depths_valid / intrinsics[1, 1]
    z_cam = depths_valid
    
    points_cam = np.stack([x_cam, y_cam, z_cam], axis=1)
    
    # Transform from camera frame to world frame
    R_cam_to_world = pose_mat[:3, :3].T
    t_cam_to_world = pose_mat[:3, 3]
    
    points_world = points_cam @ R_cam_to_world.T + t_cam_to_world
    return points_world
