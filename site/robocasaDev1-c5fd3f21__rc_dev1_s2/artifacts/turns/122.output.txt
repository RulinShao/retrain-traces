=== finish.py ===
"""python3 finish.py [--policy base | --adapters A[@VERSION][,B...]] [--note TEXT]  |  python3 finish.py --confirm TOKEN
End the session in two steps. Step one submits the policy to score (no policy arguments: the loaded stack) and the next
observation states exactly what will be scored and a token; the session continues. Step two, in a later turn,
`--confirm TOKEN` ends the session and that submission is scored on the hidden seeds. A load.py or train.py request between
the two steps cancels the submission (submit again)."""
import argparse
from _submit import submit

p = argparse.ArgumentParser(usage=__doc__)
p.add_argument("--policy", choices=["base"], help="submit the base policy (no adapters)")
p.add_argument("--adapters", help="comma-separated adapter names; single mode: current, v<k> or current@<k>")
p.add_argument("--note", help="free text kept with the submission")
p.add_argument("--confirm", metavar="TOKEN", help="step two: end the session with the pending submission")
a = p.parse_args()
if a.confirm is not None and (a.policy or a.adapters or a.note):
    p.error("--confirm takes no other arguments")
if a.policy and a.adapters:
    p.error("give either --policy base or --adapters, not both")
submit("finish", {"confirm": a.confirm} if a.confirm is not None else
       {k: v for k, v in (("policy", a.policy), ("adapters", a.adapters), ("note", a.note)) if v is not None})
=== _submit.py ===
"""Shared by train.py, eval.py, load.py, wait.py, cancel.py, finish.py: write one request file to jobs/; the environment picks it up when
the current shell command finishes, and writes jobs/<id>.json with its state and result."""
import json, os

JOBS = os.path.join(os.path.dirname(os.path.abspath(__file__)), "jobs")


def submit(kind, args):
    os.makedirs(JOBS, exist_ok=True)
    n = 1
    while True:
        try:
            f = open(os.path.join(JOBS, f"{n}.request.json"), "x")
            break
        except FileExistsError:
            n += 1
    with f:
        json.dump({"id": n, "kind": kind, "args": args}, f)
    print(f"job {n}: {kind} requested; it is picked up when this command finishes (see python3 status.py {n})")

=== unit.py ===
"""python3 unit.py register NAME --task T --reward FILE [--start-preset P] [--done FILE]  -- a subtask unit of task T: afterwards
NAME works as a task in train.py (a spec with "task": "NAME" trains on T with reward FILE, from your preset P when given),
rollout.py --task NAME (T's practice simulator, from P's states when given; rows scored by FILE and, with --done, `unit_done`) and
eval.py --task NAME (the same, counted: successes = done(r) holds, or reward(r) >= 1.0 without --done; the unit's own SCOREBOARD
row, never a final score). FILE defines reward(r) -> float, --done a file defining done(r) -> bool (both .py in your workspace,
r = the rollout row's keys); P is a preset you built with preset.py on T. Record: units/NAME.json. Nothing is charged.
python3 unit.py list  -- the session's units."""
import argparse
from _submit import submit

p = argparse.ArgumentParser(usage=__doc__)
sub = p.add_subparsers(dest="action", required=True)
r = sub.add_parser("register")
r.add_argument("name")
r.add_argument("--task", required=True)
r.add_argument("--reward", required=True, help="a .py file in your workspace defining reward(r) -> float")
r.add_argument("--start-preset", help="a preset you built with preset.py on the task")
r.add_argument("--done", help="a .py file in your workspace defining done(r) -> bool")
sub.add_parser("list")
a = p.parse_args()
args = {"action": a.action}
if a.action == "register":
    args.update(name=a.name, task=a.task, reward=a.reward, **{k: v for k, v in (("start_preset", a.start_preset), ("done", a.done)) if v})
submit("unit", args)

=== status.py ===
"""python3 status.py [ID...]  -- print the state and result of jobs (all jobs if no ID), then what finishing now would score."""
import glob, json, os, sys

if {"-h", "--help"} & set(sys.argv[1:]):
    sys.exit(__doc__)
here = os.path.dirname(os.path.abspath(__file__))
jobs = os.path.join(here, "jobs")
ids = sys.argv[1:] or sorted((os.path.basename(p)[:-5] for p in glob.glob(f"{jobs}/*.json") if ".request" not in p), key=int)
for i in ids:
    path = os.path.join(jobs, f"{i}.json")
    print(open(path).read().strip() if os.path.exists(path) else json.dumps({"id": i, "state": "requested"}))
try:   # STATUS.json is rewritten by the environment after every command
    print(json.load(open(os.path.join(here, "STATUS.json")))["finish_now"])
except (OSError, ValueError, KeyError):
    pass

"""python3 rollout.py --adapter A[,B...]|base|loaded --task T --seeds N [--reward FILE]  -- run the policy (that adapter stack) on N
fresh dev seeds of task T and return, per seed, its program, the tail of what the program printed and the run facts (seed,
sandbox_rc, secs, wall_s, task), plus task_completed and stdout_tail on a task whose verifier you can see, and there, where the
site runs the programs on the task's practice simulator, sim_reward, stages and final_state. --reward FILE (a .py in
your workspace defining reward(result) -> float) also scores each seed with your reward over those keys (not the program).
Charged as an eval of N seeds. The rows are written to rollouts/<job id>.jsonl; the job result and SCOREBOARD.md give the mean of
your reward."""

from _submit import submit
