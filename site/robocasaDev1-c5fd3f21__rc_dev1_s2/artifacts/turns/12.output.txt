# Helper function to convert pixel coordinates to 3D points
def pixel_to_3d(u, v, depth_val, intrinsics, pose_mat):
    if depth_val <= 0.05 or depth_val > 3:
        return None
    x_cam = (u - intrinsics[0, 2]) * depth_val / intrinsics[0, 0]
    y_cam = (v - intrinsics[1, 2]) * depth_val / intrinsics[1, 1]
    z_cam = depth_val
    point_cam = np.array([x_cam, y_cam, z_cam, 1.0])
    point_world = pose_mat @ point_cam
    return point_world[:3]

# Helper function to get centroid of a mask
def get_mask_centroid(mask, depth, intrinsics, pose_mat):
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return None
    # Use the median for robustness
    u = int(np.median(xs))
    v = int(np.median(ys))
    depth_val = depth[v, u]
    return pixel_to_3d(u, v, depth_val, intrinsics, pose_mat)
