# reward_check: run your reward function before you pay for training

`python3 skills/optional/reward_check/check_reward.py REWARD [--results FILE]` loads `reward(result)` from a .py file or from
your train spec (inline `def reward` source, or a .py path) and prints its score on three example sim replies (task
completed, not completed, program crashed), or on replies you provide (`--results`, a JSON list or JSONL).

`result` is the sim service's reply for one rollout: `task_completed`, `reward` (the simulator's staged reward),
`sandbox_rc`, `seed`, `secs`, `wall_s`, `log_tail`, `task` (also readable as attributes). During training an exception
in reward() fails the train job. A reward that gives every rollout the same score gives GRPO nothing to learn from.
Free: it runs locally and sends nothing.
===
"""usage: python3 skills/optional/reward_check/check_reward.py REWARD [--results FILE]

REWARD: a .py file defining reward(result), or your train spec (JSON or YAML) whose `reward` is inline `def reward`
source or a .py path. Runs reward(result) on three example sim replies (task completed / not completed / program
crashed), or on your own replies (--results: a JSON list or JSONL of dicts), and prints each score or the exception.
Nothing is sent anywhere and nothing is judged: the output is the scores.
"""
import argparse, json, os, traceback

EXAMPLES = [   # the keys of the sim service's /run reply, which is what reward(result) receives during training
    {"case": "completed", "task_completed": True, "reward": 1.0, "sandbox_rc": 0, "seed": 0, "secs": 30.0,
     "wall_s": 31.0, "log_tail": "", "task": "T"},
    {"case": "not completed", "task_completed": False, "reward": 0.25, "sandbox_rc": 0, "seed": 1, "secs": 30.0,
     "wall_s": 31.0, "log_tail": "", "task": "T"},
    {"case": "program crashed", "task_completed": False, "reward": 0.0, "sandbox_rc": 1, "seed": 2, "secs": 2.0,
     "wall_s": 3.0, "log_tail": "Traceback (most recent call last): ...", "task": "T"},
]


class Result(dict):
    """result["task_completed"] or result.task_completed, as during training."""
    __getattr__ = dict.__getitem__


def reward_source(path):
    if path.endswith(".py"):
        return open(path).read()
    text = open(path).read()
    try:
        spec = json.loads(text)
    except json.JSONDecodeError:
        import yaml
        spec = yaml.safe_load(text)
    r = spec.get("reward") if isinstance(spec, dict) else None
    if isinstance(r, str) and "def reward" in r:
        return r
    if isinstance(r, str) and r.endswith(".py"):
        return open(os.path.join(os.path.dirname(os.path.abspath(path)), r)).read()
    raise SystemExit(f"no reward source in {path}: `reward` is {r!r} (a built-in name, or missing)")


def load_results(path):
    text = open(path).read().strip()
    return json.loads(text) if text.startswith("[") else [json.loads(line) for line in text.splitlines() if line.strip()]


def main():
    p = argparse.ArgumentParser(usage=__doc__)
    p.add_argument("reward")
    p.add_argument("--results")
    a = p.parse_args()
    scope = {}
    exec(compile(reward_source(a.reward), a.reward, "exec"), scope)
    if "reward" not in scope:
        raise SystemExit("the source defines no function named reward")
    rows = load_results(a.results) if a.results else EXAMPLES
    for i, row in enumerate(rows):
        label = row.get("case", f"result {i}")
        try:
            print(f"{label}: {scope['reward'](Result(row))!r}")

"""Shared by train.py, eval.py, load.py, wait.py, cancel.py, finish.py: write one request file to jobs/; the environment picks it up when
the current shell command finishes, and writes jobs/<id>.json with its state and result."""
import json, os

JOBS = os.path.join(os.path.dirname(os.path.abspath(__file__)), "jobs")


def submit(kind, args):
    os.makedirs(JOBS, exist_ok=True)
    n = 1
    while True:
        try:
            f = open(os.path.join(JOBS, f"{n}.request.json"), "x")
            break
        except FileExistsError:
            n += 1
    with f:
        json.dump({"id": n, "kind": kind, "args": args}, f)
    print(f"job {n}: {kind} requested; it is picked up when this command finishes (see python3 status.py {n})")
