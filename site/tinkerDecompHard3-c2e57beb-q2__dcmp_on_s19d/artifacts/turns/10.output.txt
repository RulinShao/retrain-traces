# Step 5: Transform the grasp pose from camera frame to world frame
# The camera extrinsic matrix pose_mat transforms from camera frame to world frame
# So we need to multiply: T_world_cam @ T_cam_grasp
T_world_cam = pose_mat
T_world_grasp = T_world_cam @ best_grasp_pose_cam

# Extract position and orientation of the grasp in world frame
grasp_pos_world = T_world_grasp[:3, 3].copy()
grasp_quat_wxyz = np.array([1.0, 0.0, 0.0, 0.0])  # Identity quaternion initially

# Actually, let's extract the rotation from the grasp pose
R_cam_grasp = best_grasp_pose_cam[:3, :3]
R_world_cam = pose_mat[:3, :3]
R_world_grasp = R_world_cam @ R_cam_grasp

# Convert rotation matrix to quaternion (wxyz)
def rotmat_to_quat(R):
    trace = R[0, 0] + R[1, 1] + R[2, 2]
    if trace > 0:
        S = np.sqrt(trace + 1.0) * 2
        w = 0.25 * S
        x = (R[2, 1] - R[1, 2]) / S
        y = (R[0, 2] - R[2, 0]) / S
        z = (R[1, 0] - R[0, 1]) / S
    elif R[0, 0] > R[1, 1] and R[0, 0] > R[2, 2]:
        S = np.sqrt(1.0 + R[0, 0] - R[1, 1] - R[2, 2]) * 2
        w = (R[2, 1] - R[1, 2]) / S
        x = 0.25 * S
        y = (R[0, 1] + R[1, 0]) / S
        z = (R[0, 2] + R[2, 0]) / S
    elif R[1, 1] > R[2, 2]:
        S = np.sqrt(1.0 + R[1, 1] - R[0, 0] - R[2, 2]) * 2
        w = (R[0, 2] - R[2, 0]) / S
        x = (R[0, 1] + R[1, 0]) / S
        y = 0.25 * S
        z = (R[1, 2] + R[2, 1]) / S
    else:
        S = np.sqrt(1.0 + R[2, 2] - R[0, 0] - R[1, 1]) * 2
        w = (R[1, 0] - R[0, 1]) / S
        x = (R[0, 2] + R[2, 0]) / S
        y = (R[1, 2] + R[2, 1]) / S
        z = 0.25 * S
    quat = np.array([w, x, y, z])
    return quat / np.linalg.norm(quat)

grasp_quat_wxyz = rotmat_to_quat(R_world_grasp)

# Step 7: Compute the center of the red cube and its extent
ys_r, xs_r = np.where(red_mask)
if len(xs_r) == 0:
    raise ValueError("Red cube mask is empty.")

xs_r_arr = xs_r.astype(np.float64)
ys_r_arr = ys_r.astype(np.float64)
zs_r_arr = depth[ys_r, xs_r].astype(np.float64)

valid_r = zs_r_arr > 0
xs_r_arr = xs_r_arr[valid_r]
ys_r_arr = ys_r_arr[valid_r]
zs_r_arr = zs_r_arr[valid_r]

if len(xs_r_arr) == 0:
    raise ValueError("No valid depth points for red cube.")

X_cam_r = (xs_r_arr - cx) * zs_r_arr / fx
Y_cam_r = (ys_r_arr - cy) * zs_r_arr / fy
Z_cam_r = zs_r_arr

points_cam_r = np.column_stack((X_cam_r, Y_cam_r, Z_cam_r))
points_hom_r = np.hstack((points_cam_r, np.ones((len(points_cam_r), 1))))
points_world_r = (pose_mat @ points_hom_r.T).T[:, :3]

red_extent = obb_red["extent"]
