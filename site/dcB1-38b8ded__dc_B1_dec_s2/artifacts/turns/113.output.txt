    p_world = cam_to_world @ p_cam + t
    return p_world

# Sample multiple points across the bounding box to estimate world extents
sample_points = []
for i in range(5):
    for j in range(5):
        px = int(x_min_px + (x_max_px - x_min_px) * i / 4)
        py = int(y_min_px + (y_max_px - y_min_px) * j / 4)
        d = depth[py, px]
        if d > 0.01:  # valid depth
            p_w = unproject_pixel_to_world(px, py, d, intrinsics, pose_mat)
            sample_points.append(p_w)

if not sample_points:
    raise ValueError("No valid depth points found for spill.")

sample_points = np.array(sample_points)
x_min_w = sample_points[:, 0].min()
x_max_w = sample_points[:, 0].max()
y_min_w = sample_points[:, 1].min()
y_max_w = sample_points[:, 1].max()

# Step 5: Define wiping trajectory at z=0.0
z_wipe = 0.0
quaternion_wxyz = np.array([1.0, 0.0, 0.0, 0.0])  # downward-facing orientation

# Create small wiping motions within the spill bounds
n_steps_x = 3
n_steps_y = 3

wipe_positions = []
for i in range(n_steps_x):
    for j in range(n_steps_y):
        x = x_min_w + (x_max_w - x_min_w) * i / (n_steps_x - 1) if n_steps_x > 1 else (x_min_w + x_max_w)/2
        y = y_min_w + (y_max_w - y_min_w) * j / (n_steps_y - 1) if n_steps_y > 1 else (y_min_w + y_max_w)/2
        wipe_positions.append(np.array([x, y, z_wipe]))

# Add intermediate points to avoid large deltas
all_positions = []
for k in range(len(wipe_positions)):
    if k == 0:
        all_positions.append(wipe_positions[k])
    else:
        prev = all_positions[-1]
        curr = wipe_positions[k]
        # Interpolate between previous and current with small steps
        num_interp = 5
        for t in np.linspace(0, 1, num_interp)[1:]:
            interp_pos = prev + t * (curr - prev)
            all_positions.append(interp_pos)

# Step 6: Execute wiping motion
for pos in all_positions:
    joints = solve_ik(pos, quaternion_wxyz)
    move_to_joints(joints)
